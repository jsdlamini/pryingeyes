% Stick-figure actor. \actor{name}{(x,y)}{label}; a coordinate `name` sits at the chest.
\newcommand{\actor}[3]{%
  \begin{scope}[shift={#2}]
    \draw[ink,line width=0.75pt,fill=white] (0,1.0) circle (0.17);
    \draw[ink,line width=0.75pt] (0,0.83)--(0,0.38) (-0.3,0.66)--(0.3,0.66) (0,0.38)--(-0.24,0) (0,0.38)--(0.24,0);
    \node[font=\sffamily\scriptsize\bfseries,text=ink,align=center,anchor=north] at (0,-0.04) {#3};
    \coordinate (#1) at (0,0.6);
  \end{scope}}
